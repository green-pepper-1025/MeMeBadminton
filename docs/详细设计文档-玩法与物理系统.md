### 2.2 羽毛球物理模型

#### 核心参数
```typescript
interface BallPhysicsConfig {
    // 基础属性
    radius: 20;                 // 半径 (px)
    mass: 1;                    // 质量 (kg)
    
    // 材质属性
    restitution: 0.6;           // 弹性系数（反弹高度）
    friction: 0.1;              // 摩擦系数
    
    // 速度限制
    maxSpeed: 1500;             // 最大速度 (px/s)
    minSpeed: 50;               // 最小速度（低于此值视为静止）
    
    // 阻尼系数
    linearDamping: 0.8;         // 线性阻尼（模拟空气阻力）
    angularDamping: 0.5;        // 角阻尼（旋转减速）
}
```

#### 飞行轨迹计算

**参考 StickBadminton 的物理特性**:
1. **抛物线轨迹**: 受重力影响的标准抛物线
2. **速度衰减**: 线性阻尼模拟羽毛球减速
3. **反弹衰减**: 每次反弹损失 40% 能量

```typescript
class BallPhysics {
    /**
     * 更新球的物理状态（每帧调用）
     */
    update(deltaTime: number): void {
        // 1. 应用重力
        this.velocity.y += PHYSICS_CONFIG.gravity * 
                           PHYSICS_CONFIG.gravityScale * 
                           deltaTime;
        
        // 2. 应用线性阻尼（空气阻力）
        const dampingFactor = Math.pow(1 - this.config.linearDamping, deltaTime);
        this.velocity.x *= dampingFactor;
        this.velocity.y *= dampingFactor;
        
        // 3. 限制速度
        const speed = this.velocity.length();
        if (speed > this.config.maxSpeed) {
            this.velocity.normalize().multiplyScalar(this.config.maxSpeed);
        }
        
        // 4. 更新位置
        this.position.x += this.velocity.x * deltaTime;
        this.position.y += this.velocity.y * deltaTime;
    }
    
    /**
     * 处理地面碰撞
     */
    onGroundCollision(): void {
        // 反弹速度 = 原速度 * 弹性系数
        this.velocity.y = -this.velocity.y * this.config.restitution;
        
        // 水平速度衰减（摩擦力）
        this.velocity.x *= (1 - this.config.friction);
        
        // 检查是否静止
        if (Math.abs(this.velocity.y) < this.config.minSpeed) {
            this.velocity.y = 0;
            this.isGrounded = true;
        }
    }
}
```

### 2.3 击球物理

#### 击球力度计算

**参考 StickBadminton**: 击球力度取决于球拍挥动速度和击球角度

```typescript
interface HitCalculation {
    /**
     * 计算击球后的球速
     * @param racketVelocity 球拍速度向量
     * @param racketAngle 球拍角度（弧度）
     * @param ballVelocity 球当前速度
     */
    calculateHitVelocity(
        racketVelocity: Vec2, 
        racketAngle: number, 
        ballVelocity: Vec2
    ): Vec2 {
        // 1. 计算球拍法向量（垂直于球拍表面）
        const racketNormal = new Vec2(
            Math.cos(racketAngle + Math.PI / 2),
            Math.sin(racketAngle + Math.PI / 2)
        );
        
        // 2. 计算球拍速度在法向量上的投影
        const racketSpeed = racketVelocity.dot(racketNormal);
        
        // 3. 基础击球速度 = 球拍速度 * 力度倍率
        const baseHitSpeed = racketSpeed * HIT_POWER_MULTIPLIER;
        
        // 4. 叠加球的原有速度（部分保留）
        const resultVelocity = racketNormal
            .multiplyScalar(baseHitSpeed)
            .add(ballVelocity.multiplyScalar(0.3));
        
        return resultVelocity;
    }
}

// 常量定义
const HIT_POWER_MULTIPLIER = 1.5;  // 击球力度倍率
const SMASH_ANGLE_THRESHOLD = -45; // 扣杀角度阈值（度）
const SMASH_POWER_BONUS = 1.3;     // 扣杀力度加成
```

#### 击球类型判定

```typescript
enum HitType {
    NORMAL = 'normal',      // 普通击球
    SMASH = 'smash',        // 扣杀（向下击球）
    LOB = 'lob',            // 高远球（向上击球）
    DROP = 'drop'           // 吊球（轻击）
}

function determineHitType(racketAngle: number, hitSpeed: number): HitType {
    const angleDeg = racketAngle * 180 / Math.PI;
    
    if (angleDeg < SMASH_ANGLE_THRESHOLD && hitSpeed > 800) {
        return HitType.SMASH;
    } else if (angleDeg > 60 && hitSpeed > 600) {
        return HitType.LOB;
    } else if (hitSpeed < 300) {
        return HitType.DROP;
    }
    return HitType.NORMAL;
}
```

### 2.4 碰撞检测详细设计

#### 碰撞事件处理流程

```typescript
class CollisionHandler {
    /**
     * 球与球拍碰撞
     */
    onBallRacketCollision(ball: Ball, racket: Racket, player: Player): void {
        // 1. 记录击球玩家
        ball.lastHitBy = player.id;
        
        // 2. 计算击球速度
        const racketVelocity = racket.getVelocity();
        const racketAngle = racket.getAngle();
        const newVelocity = this.calculateHitVelocity(
            racketVelocity, 
            racketAngle, 
            ball.velocity
        );
        
        // 3. 应用新速度
        ball.setVelocity(newVelocity);
        
        // 4. 判定击球类型
        const hitType = determineHitType(racketAngle, newVelocity.length());
        
        // 5. 触发事件
        EventBus.emit(GameEvent.PLAYER_HIT_BALL, {
            playerId: player.id,
            hitType: hitType,
            velocity: newVelocity,
            position: ball.position
        });
        
        // 6. 播放音效
        AudioManager.play(`hit_${hitType}`);
        
        // 7. 技能充能
        player.skillSystem.addCharge(SKILL_CHARGE_PER_HIT);
    }
    
    /**
     * 球与地面碰撞
     */
    onBallGroundCollision(ball: Ball, groundSide: 'left' | 'right'): void {
        // 1. 判定得分
        const scoringPlayer = (groundSide === 'left') ? 'player2' : 'player1';
        
        // 2. 检查是否为有效得分
        if (ball.lastHitBy !== scoringPlayer) {
            EventBus.emit(GameEvent.SCORE_CHANGED, {
                scoringPlayer: scoringPlayer,
                reason: 'ball_landed',
                position: ball.position
            });
        }
        
        // 3. 物理反弹
        ball.onGroundCollision();
    }
    
    /**
     * 球与球网碰撞
     */
    onBallNetCollision(ball: Ball): void {
        // 1. 反弹处理
        ball.velocity.x = -ball.velocity.x * 0.5;
        ball.velocity.y *= 0.8;
        
        // 2. 检查是否触网落地（违规）
        if (ball.position.y < NET_TOP_Y) {
            EventBus.emit(GameEvent.BALL_OUT, {
                reason: 'net_violation',
                lastHitBy: ball.lastHitBy
            });
        }
    }
}
```

### 3.2 挥拍动画系统

```typescript
class SwingAnimation {
    private _isSwinging: boolean = false;
    private _swingDuration: number = 0.2;  // 挥拍持续时间
    private _swingTimer: number = 0;
    
    /**
     * 触发挥拍
     */
    swing(): void {
        if (this._isSwinging) return;
        
        this._isSwinging = true;
        this._swingTimer = 0;
        
        // 播放挥拍动画
        this._playSwingAnimation();
        
        // 启用击球碰撞检测
        this._enableHitDetection();
    }
    
    /**
     * 更新挥拍状态
     */
    update(deltaTime: number): void {
        if (!this._isSwinging) return;
        
        this._swingTimer += deltaTime;
        
        if (this._swingTimer >= this._swingDuration) {
            this._isSwinging = false;
            this._disableHitDetection();
        }
    }
    
    /**
     * 播放挥拍动画（球拍快速旋转）
     */
    private _playSwingAnimation(): void {
        // 球拍从当前角度快速旋转 90 度
        const startAngle = this._racket.angle;
        const endAngle = startAngle + Math.PI / 2;
        
        tween(this._racket)
            .to(this._swingDuration, { angle: endAngle })
            .start();
    }
}
```

## 五、得分与比赛系统

### 5.1 得分规则

```typescript
enum ScoreReason {
    BALL_LANDED = 'ball_landed',        // 球落地
    BALL_OUT = 'ball_out',              // 球出界
    NET_VIOLATION = 'net_violation',    // 触网违规
    DOUBLE_HIT = 'double_hit'           // 连击违规（可选）
}

class ScoreManager {
    /**
     * 判定得分
     */
    checkScore(ball: Ball, reason: ScoreReason): string | null {
        switch (reason) {
            case ScoreReason.BALL_LANDED:
                // 球落在对方半场 → 得分
                const landSide = ball.position.x < 0 ? 'left' : 'right';
                return landSide === 'left' ? 'player2' : 'player1';
            
            case ScoreReason.BALL_OUT:
                // 球出界 → 最后击球者失分
                return ball.lastHitBy === 'player1' ? 'player2' : 'player1';
            
            case ScoreReason.NET_VIOLATION:
                // 触网 → 最后击球者失分
                return ball.lastHitBy === 'player1' ? 'player2' : 'player1';
        }
    }
}
```

### 5.2 比赛流程

```typescript
class MatchManager {
    private _player1Score: number = 0;
    private _player2Score: number = 0;
    private _player1RoundsWon: number = 0;
    private _player2RoundsWon: number = 0;
    private _currentRound: number = 1;
    
    /**
     * 更新得分
     */
    addScore(playerId: string): void {
        if (playerId === 'player1') {
            this._player1Score++;
        } else {
            this._player2Score++;
        }
        
        // 检查回合结束
        if (this._player1Score >= 11 || this._player2Score >= 11) {
            this._endRound();
        }
    }
    
    /**
     * 结束回合
     */
    private _endRound(): void {
        const winner = this._player1Score > this._player2Score ? 'player1' : 'player2';
        
        if (winner === 'player1') {
            this._player1RoundsWon++;
        } else {
            this._player2RoundsWon++;
        }
        
        // 检查比赛结束（三局两胜）
        if (this._player1RoundsWon >= 2 || this._player2RoundsWon >= 2) {
            this._endMatch();
        } else {
            this._startNewRound();
        }
    }
}
```

## 六、技能系统设计

### 6.1 充能机制

```typescript
class SkillCharger {
    private _currentCharge: number = 0;
    private _maxCharge: number = 100;
    private _usesRemaining: number = 3;
    
    // 充能速率
    private readonly TIME_CHARGE_RATE = 100 / 20;  // 20秒充满
    private readonly HIT_CHARGE_AMOUNT = 15;       // 每次击球充能15%
    
    /**
     * 时间充能（每帧调用）
     */
    updateTimeCharge(deltaTime: number): void {
        if (this._currentCharge < this._maxCharge) {
            this._currentCharge += this.TIME_CHARGE_RATE * deltaTime;
            this._currentCharge = Math.min(this._currentCharge, this._maxCharge);
            
            if (this._currentCharge >= this._maxCharge) {
                EventBus.emit(GameEvent.PLAYER_SKILL_READY, { playerId: this._playerId });
            }
        }
    }
    
    /**
     * 击球充能
     */
    onHitBall(): void {
        this._currentCharge += this.HIT_CHARGE_AMOUNT;
        this._currentCharge = Math.min(this._currentCharge, this._maxCharge);
    }
    
    /**
     * 使用技能
     */
    useSkill(): boolean {
        if (this._currentCharge < this._maxCharge || this._usesRemaining <= 0) {
            return false;
        }
        
        this._currentCharge = 0;
        this._usesRemaining--;
        return true;
    }
}
```
